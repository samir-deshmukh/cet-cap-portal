/* ── getHistory(college) ─────────────────────────────────────
   Returns the 4-year percentile history for a college.
   If the college has its own `history` array, use that.
   Otherwise, auto-generate one from its current percentile
   plus the offsets defined in CONFIG.historyOffsets.          */
function getHistory(college) {
  if (Array.isArray(college.history) && college.history.length) {
    return college.history;
  }
  const years   = CONFIG.historyYears;
  const offsets = CONFIG.historyOffsets;
  return years.map((y, i) => ({
    year: y,
    percentile: Math.max(0, Math.min(100, college.percentile + (offsets[i] ?? 0)))
  }));
}

/* ── buildGapChart(college, userPct) ────────────────────────
   Builds the HTML for the "gap vs cutoff across years" chart.
   For each year, gap = userPct − thatYearCutoff.
   Positive gaps draw bars ABOVE the zero line (dark).
   Negative gaps draw bars BELOW the zero line (light grey).
   The tallest bar fills the whole half-height; others scale
   proportionally against maxAbs.                              */
function buildGapChart(college, userPct) {
  /* IMPORTANT: this chart has its own data source. Do NOT use
     percentileRange(), c.low/c.high, or any all-years eligibility range.
     graph_history contains the raw yearly MIN/MAX percentile pair for
     this exact college+course and exact year. */
  const rawGraphHistory = Array.isArray(college.graph_history)
    ? college.graph_history
    : [];

  const yearRanges = rawGraphHistory
    .map(h => {
      const year = Number(h.year);
      const low = Number(h.low);
      const high = Number(h.high);
      if (!Number.isFinite(year) || !Number.isFinite(low) || !Number.isFinite(high)) return null;
      return {
        year,
        low: Math.max(0, Math.min(100, Math.min(low, high))),
        high: Math.max(0, Math.min(100, Math.max(low, high)))
      };
    })
    .filter(Boolean)
    .sort((a, b) => a.year - b.year);

  const highest = yearRanges.reduce((a, b) => !a || b.high > a.high ? b : a, null);
  const lowest = yearRanges.reduce((a, b) => !a || b.low < a.low ? b : a, null);

  const formatPct = value => {
    const n = Number(value);
    if (!Number.isFinite(n)) return "—";
    return n.toFixed(2).replace(/\.?0+$/, "");
  };

  const cols = yearRanges.map(r => {
    /* Bar height is THIS YEAR'S high percentile only.
       Label is THIS YEAR'S high-low pair. */
    const rangeLabel = `${formatPct(r.high)}-${formatPct(r.low)}`;
    return `
      <div class="d-gap-col">
        <div class="d-gap-bar-wrap">
          <div class="d-gap-bar" data-h="${r.high}">
            <span class="d-gap-val">${rangeLabel}</span>
          </div>
        </div>
      </div>`;
  }).join("");

  const yearsHTML = yearRanges.map(r => `<div>${r.year}</div>`).join("");

  return `
    <div class="d-section">
      <div class="d-section-title">Cutoff Trend · Across Years</div>
      <div class="d-gap-summary">
        <span class="d-gap-summary-item hi">Highest: <strong>${highest ? formatPct(highest.high) : "—"}%</strong> (${highest ? highest.year : "—"})</span>
        <span class="d-gap-summary-item lo">Lowest: <strong>${lowest ? formatPct(lowest.low) : "—"}%</strong> (${lowest ? lowest.year : "—"})</span>
      </div>
      <div class="d-gap-chart">
        <div class="d-gap-plot">
          <div class="d-gap-grid"></div>
          <div class="d-gap-yaxis"><span>100</span><span>75</span><span>50</span><span>25</span><span>0</span></div>
          <div class="d-gap-cols">${cols}</div>
        </div>
        <div class="d-gap-years">${yearsHTML}</div>
      </div>
    </div>`;
}

/* ── renderSlide(college, userPct) ───────────────────────────
   Builds the full HTML for the slide's body: the gap chart is now
   the FIRST thing shown (nothing above it — the verdict banner and
   stat grid were removed as misleading), then about text,
   highlights, placement, admission note, and CTA buttons.          */
function renderSlide(c, userPct) {
  /* Highlight chips (only rendered if college.highlights exists) */
  const highlights = (c.highlights || []).map(h => `<span class="d-chip">${h}</span>`).join("");

  /* Placement bars (only if college.placement exists) */
  const placement = c.placement ? `
    <div class="d-section">

        </div>
        <div class="d-bar-row">
          <span class="lbl">Average CTC</span>
          <div class="d-bar-track"><div class="d-bar-fill" data-w="65"></div></div>
          <span class="val">${c.placement.avg}</span>
        </div>
        <div class="d-bar-row">
          <span class="lbl">Highest CTC</span>
          <div class="d-bar-track"><div class="d-bar-fill" data-w="100"></div></div>
          <span class="val">${c.placement.highest}</span>
        </div>
      </div>
    </div>` : "";

  const about = c.about ? `
    <div class="d-section">
      <div class="d-section-title">About the Programme</div>
      <p class="d-text">${c.about}</p>
    </div>` : "";

  const note = c.eligibilityNote ? `
    <div class="d-section">
      <div class="d-section-title">Admission Note</div>
      <p class="d-text">${c.eligibilityNote}</p>
    </div>` : "";

  /* Return everything as one big HTML string */
  return `
    ${buildGapChart(c, userPct)}
    ${renderDrawerTrendSection()}
    ${about}

    <div class="d-cta-row">
      ${c.website ? `<a class="d-cta primary" href="${c.website}" target="_blank" rel="noopener">Visit Website ↗</a>` : ""}
    </div>
  `;
}

/* ═══════════════════════════════════════════════════════════════════
   ██  enhanceSelect(selectEl, opts) — searchable-dropdown wrapper
   ───────────────────────────────────────────────────────────────────
   Wraps a plain, already-populated <select> with a custom button +
   panel UI, WITHOUT changing how the rest of the app talks to it: the
   native <select> stays in the DOM (hidden), still fires "change",
   still has a normal .value and .disabled — every existing piece of
   code (populateSelect(), citySelect.value, etc.) keeps working
   completely unmodified. The panel is always the same fixed size and
   its option list scrolls once there are more items than fit, and
   opts.search adds a live-filter search box above the list (the full
   option list is still shown underneath, just narrowed by the
   filter — nothing is ever removed from the underlying select). ── */
function enhanceSelect(selectEl, opts = {}) {
  if (!selectEl || selectEl._enhanced) return selectEl && selectEl._enhanced;
  const search = opts.search !== false;
